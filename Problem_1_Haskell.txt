import Text.Printf (printf)
import System.Exit (exitSuccess)

-- getValidInput for input
getValidInput :: IO [Int]
getValidInput = do
    line <- getLine
    let nums = map read (words line) :: [Int]
    
    -- checks if integer is out of bounds
    -- If invalid terminate
    if length nums == 10 && all (\x -> x >= 1 && x <= 100) nums
        then return nums
        else exitSuccess

main :: IO ()
main = do
    nums <- getValidInput
    
    -- extraction of 7th number
    let seventh_number = nums !! 6
    
    -- calculate  of allowance
    let weekly_allowance :: Double
        weekly_allowance = fromIntegral seventh_number * 2.75
    
    printf "The multiplier is %d! Lulu's allowance for the week is $ %.2f!\n" seventh_number weekly_allowance
