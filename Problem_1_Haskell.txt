import Text.Printf (printf)

main :: IO ()
main = do
    lineInput <- getLine
    
    -- split the line into words and convert to double
    let numbers = map read (words lineInput) :: [Double]
    
    if length numbers >= 10
        then do
            let multiplier = numbers !! 6
            
            -- constraints
            if multiplier >= 1 && multiplier <= 100
                then do
                    let allowance = multiplier * 2.75
                    printf "The multiplier is %.0f! Lulu's allowance for the week is $ %.2f!\n" multiplier allowance
                else return ()
        else return ()
