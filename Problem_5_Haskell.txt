main :: IO ()
main = do
    line <- getLine
    let allParts = words line
    
    case allParts of
        --head tail logic
        (h:t) -> do
            let intN = read h :: Int
            -- take only the first intN elements from the tail
            let intn = take intN (map read t :: [Integer])
            
            if 2 <= intN && intN <= 100 && all (\n -> n >= -1000 && n <= 1000) intn
                then do
                    let evensReversed = filter even (reverse intn)
                    putStrLn (unwords (map show evensReversed))
                else return ()
        
        -- if user has no input
        [] -> return ()
