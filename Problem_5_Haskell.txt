import System.Exit (exitSuccess)

main :: IO ()
main = do
    nStr <- getLine
    let n = read nStr :: Int
    
    if n < 2 || n > 100 
        then putStrLn "Out of bounds" >> exitSuccess
        else do
            input <- getLine
            let values = map read (words input) :: [Int]
            
            if any (\x -> x < -1000 || x > 1000) values
                then putStrLn "Out of bounds" >> exitSuccess
                else do
                    -- reverse, filter even, print
                    let evensReversed = filter even (reverse values)
                    putStrLn $ unwords (map show evensReversed)
-- input N then enter, values should be inputted in next line
