    format('The multiplier is ~w! Lulu\'s allowance for the week is $ ~2f!', [Seventh_number, Weekly_Allowance]).

count_uppercase([Head|Tail], Count) :-
    /* checks if H is A-Z */
    char_type(Head, upper),   

    /* go look at the rest of the list */
    count_uppercase(Tail, RestCount), 
    
    /* add 1 to whatever we found in the rest */
    Count is RestCount + 1.        
    
count_uppercase([Head|Tail], Count) :-
    
    /* \+ means "NOT" */
    \+ char_type(Head, upper),   
    
    /* count stays exactly the same as the Tail's count */
    count_uppercase(Tail, Count).

PROGRAM countUpper
    IMPLICIT NONE
        INTEGER :: N, t, i, j, upperCount ! declaring of variables
        CHARACTER(len=100) :: lineOfCh ! max length of 100 !

        READ *, N
        IF (N < 1 .OR. N > 100) STOP 

        
        DO i = 1, N ! loop through each test case !
            upperCount = 0
            READ *, t, lineOfCh ! reads the count and the whole string at once !
            
            IF (t < 1 .OR. t > 100) STOP
            
            DO j = 1, t ! loop through the first 't' characters of the string !
                IF (lineOfCh(j:j) >= 'A' .AND. lineOfCh(j:j) <= 'Z') THEN
                    upperCount = upperCount + 1 ! counter for uppercases in the lineof Ch !
                END IF
            END DO
            
            PRINT '(I0)', upperCount !(I0) removes leading spaces from output !
        END DO
END PROGRAM countUpper

PROGRAM evenFromEnd
    IMPLICIT NONE
    
    INTEGER :: N, i 
    INTEGER, DIMENSION(100) :: myList
    
    READ *, N, myList(1:N) ! allows to enter N and the numbers to be put inside the list inputted in one line !
    
    IF (N < 2 .OR. N > 100) STOP
    
    DO i = 1, N
        IF (-1000 > myList(i) .OR. myList(i) > 1000) STOP
    END DO
    
    
    DO i = N, 1, -1
        IF (MOD(myList(i), 2) == 0) THEN
            WRITE(*, '(I0, 1X)', ADVANCE='NO') myList(i) ! allows to print the output in one line, separated by single space !
        END IF    
    END DO
    

END PROGRAM evenFromEnd
