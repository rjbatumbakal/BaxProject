import Text.Printf (printf)
calculateAverage :: Int -> Int -> Int -> Maybe Double
calculateAverage a b c
    | a >= 1 && a <= 20 && b >= 1 && b <= 20 && c >= 1 && c <= 20 = 
        Just (fromIntegral (a + b + c) / 3.0)
    | otherwise = Nothing -- returns 'Nothing' if out of bounds

main :: IO ()
main = do
    line <- getLine
    let nums = map read (words line) :: [Int]
    -- Pattern matching 
    case nums of
        [a, b, c] -> case calculateAverage a b c of
            Just avg -> printf "%.2f\n" avg
            Nothing  -> return ()
        _ -> return () -- fail if we don't get exactly 3 numbers
-- input of three numbers must be a one liner
