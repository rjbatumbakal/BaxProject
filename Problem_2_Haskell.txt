import Data.Char (isUpper)
import Control.Monad (replicateM_)

countUpper :: String -> Int
countUpper [] = 0

-- head and tail logic
countUpper (h:t) 

    -- if h is upper, add 1 and keep going
    | isUpper h = 1 + countUpper t 
    
    -- if not, just keep going with the tail
    | otherwise = countUpper t      


processLine :: IO ()
processLine = do

    linetch<- getLine
    let parts = words linetch
    
    case parts of

        -- pattern matching, match t as first item, and followng ch as second
        (tString:ch:_) -> do  
            let t = read tString :: Int
            if t >= 1 && t <= 100
            
                -- take the first t letters from ch and count uppercase
                then print (countUpper (take t ch))
                else return ()
        _ -> return ()

main :: IO ()
main = do

    lineN <- getLine
    let n = read lineN :: Int
    
    --constraints
    if n >= 1 && n <= 100
        then replicateM_ n processLine -- loop
    else return ()
