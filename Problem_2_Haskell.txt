import Data.Char (isUpper)
import Control.Monad (replicateM_)

processLine :: IO ()
processLine = do
    linetch <- getLine
    case words linetch of
        (tStr:str:_) -> do
            let t = read tStr
            if t >= 1 && t <= 100
                {- Count uppercase in first t chars -}
                then print $ length $ filter isUpper (take t str) 
                else return ()
        _ -> return () 

main :: IO ()
main = do
    lineN <- getLine
    {- read total number of test cases -}
    let n = read lineN
    {- constraints -}
    if n >= 1 && n <= 100
        {- loop -}
        then replicateM_ n processLine
        else return ()
